using Dapper;
using Microsoft.Data.SqlClient;
using TatumWelcome2.BLL.IServices;
using TatumWelcome2.BLL.Services;
using TatumWelcome2.Domain.Interfaces;
using TatumWelcome2.Domain.Models;
using TatumWelcome2.Infrastructure.Database;

namespace TatumWelcome2.Infrastructure.Repositories;

/// <summary>
/// Accounts and their ledger. Money-moving methods lock the affected rows
/// (SELECT ... FOR UPDATE) and apply only the checks that must hold under that lock: PIN
/// against the stored hash, frozen status, available funds and the balance
/// cap. Input validation, references and descriptions come from the caller.
/// </summary>
public sealed class AccountRepository : IAccountRepository
{
    private const string InsertTransactionSql = """
        INSERT INTO Transactions
            (AccountNumber, Type, Amount, Date,
             Description, Reference)
        VALUES
            (@AccountNumber, @Type, @Amount, @Date,
             @Description, @Reference);
        """;

    private const string CreditSql = """
        UPDATE Accounts
        SET Balance = Balance + @Amount
        WHERE AccountNumber = @AccountNumber;
        """;

    private const string DebitSql = """
        UPDATE Accounts
        SET Balance = Balance - @Amount
        WHERE AccountNumber = @AccountNumber;
        """;

    private readonly IDbConnectionFactory _connectionFactory;
    private readonly IPinHasher _pinHasher;

    public AccountRepository(
        IDbConnectionFactory connectionFactory,
        IPinHasher pinHasher)
    {
        _connectionFactory = connectionFactory;
        _pinHasher = pinHasher;
    }

    public Account? FindByAccountNumber(string accountNumber)
    {
        using var connection = _connectionFactory.CreateConnection();

        const string accountSql = """
            SELECT AccountNumber, Name, Branch, Phone,
                   PinHash, Balance, IsFrozen
            FROM Accounts
            WHERE AccountNumber = @AccountNumber;
            """;

        var row = connection.QuerySingleOrDefault<AccountRow>(
            accountSql,
            new { AccountNumber = accountNumber });

        if (row == null)
            return null;

        const string historySql = """
            SELECT Type, Amount, Date, Description, Reference
            FROM Transactions
            WHERE AccountNumber = @AccountNumber
            ORDER BY TransactionId;
            """;

        var history = connection.Query<TransactionRow>(
            historySql,
            new { AccountNumber = accountNumber });

        List<Transaction> transactions = history.Select(item =>
            new Transaction
            {
                Type = Enum.Parse<TransactionType>(item.Type),
                Amount = item.Amount,
                Date = item.Date,
                Description = item.Description,
                Reference = item.Reference
            }).ToList();

        return new Account
        {
            AccountNumber = row.AccountNumber,
            Name = row.Name,
            Branch = row.Branch,
            Phone = row.Phone,
            PinHash = row.PinHash,
            Balance = row.Balance,
            IsFrozen = row.IsFrozen,
            Transactions = transactions
        };
    }

    public AccountCreationResult CreateAccount(NewAccount account)
    {
        using var connection = _connectionFactory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            connection.Execute(
                """
                INSERT INTO Accounts
                    (AccountNumber, Name, Branch, PinHash,
                     Phone, Balance, IsFrozen)
                VALUES
                    (@AccountNumber, @Name, @Branch, @PinHash,
                     @Phone, @Balance, @IsFrozen);
                """,
                new
                {
                    account.AccountNumber,
                    account.Name,
                    account.Branch,
                    account.PinHash,
                    account.Phone,
                    Balance = account.OpeningDeposit,
                    IsFrozen = false
                },
                transaction);

            InsertTransaction(
                connection,
                transaction,
                account.AccountNumber,
                TransactionType.Deposit,
                account.OpeningDeposit,
                DateTime.Now,
                account.OpeningDepositDescription,
                account.OpeningDepositReference);

            transaction.Commit();

            return AccountCreationResult.Success;
        }
        catch (SqlException ex)
            when (SqlServerErrors.IsUniqueViolation(ex, "UQ_Accounts_Phone"))
        {
            transaction.Rollback();
            return AccountCreationResult.DuplicatePhone;
        }
        catch (SqlException ex)
            when (SqlServerErrors.IsUniqueViolation(ex, "PK_Accounts"))
        {
            transaction.Rollback();
            return AccountCreationResult.DuplicateAccountNumber;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public TransactionResult Deposit(
        string accountNumber,
        decimal amount,
        string reference,
        string description)
    {
        using var connection = _connectionFactory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            LockedAccountRow? account = LockAccount(connection, transaction, accountNumber);

            if (account == null)
                return Abort(transaction, TransactionResult.AccountNotFound);

            if (account.IsFrozen)
                return Abort(transaction, TransactionResult.AccountFrozen);

            if (account.Balance > AccountLimits.MaximumBalance - amount)
                return Abort(transaction, TransactionResult.BalanceLimitExceeded);

            connection.Execute(
                CreditSql,
                new { AccountNumber = accountNumber, Amount = amount },
                transaction);

            InsertTransaction(
                connection,
                transaction,
                accountNumber,
                TransactionType.Deposit,
                amount,
                DateTime.Now,
                description,
                reference);

            transaction.Commit();
            return TransactionResult.Success;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public TransactionResult Withdraw(
        string accountNumber,
        decimal amount,
        string pin,
        string reference,
        string description)
    {
        using var connection = _connectionFactory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            LockedAccountRow? account = LockAccount(connection, transaction, accountNumber);

            if (account == null)
                return Abort(transaction, TransactionResult.AccountNotFound);

            if (!_pinHasher.Verify(pin, account.PinHash))
                return Abort(transaction, TransactionResult.InvalidPin);

            if (account.IsFrozen)
                return Abort(transaction, TransactionResult.AccountFrozen);

            if (amount > account.Balance)
                return Abort(transaction, TransactionResult.InsufficientFunds);

            connection.Execute(
                DebitSql,
                new { AccountNumber = accountNumber, Amount = amount },
                transaction);

            InsertTransaction(
                connection,
                transaction,
                accountNumber,
                TransactionType.Withdrawal,
                amount,
                DateTime.Now,
                description,
                reference);

            transaction.Commit();
            return TransactionResult.Success;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public TransactionResult Transfer(
        string senderNumber,
        string recipientNumber,
        decimal amount,
        string pin,
        string reference)
    {
        using var connection = _connectionFactory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            // Both rows locked in key order to avoid deadlocks between
            // opposite-direction transfers.
            List<LockedAccountRow> rows = connection.Query<LockedAccountRow>(
                """
                SELECT AccountNumber, Name, PinHash, Balance, IsFrozen
                FROM Accounts
                WHERE AccountNumber IN (@SenderNumber, @RecipientNumber)
                ORDER BY AccountNumber
                FOR UPDATE;
                """,
                new
                {
                    SenderNumber = senderNumber,
                    RecipientNumber = recipientNumber
                },
                transaction).ToList();

            LockedAccountRow? sender = rows.FirstOrDefault(
                row => row.AccountNumber == senderNumber);

            LockedAccountRow? recipient = rows.FirstOrDefault(
                row => row.AccountNumber == recipientNumber);

            if (sender == null)
                return Abort(transaction, TransactionResult.AccountNotFound);

            if (recipient == null)
                return Abort(transaction, TransactionResult.InvalidRecipient);

            if (!_pinHasher.Verify(pin, sender.PinHash))
                return Abort(transaction, TransactionResult.InvalidPin);

            if (sender.IsFrozen)
                return Abort(transaction, TransactionResult.AccountFrozen);

            if (amount > sender.Balance)
                return Abort(transaction, TransactionResult.InsufficientFunds);

            if (recipient.Balance > AccountLimits.MaximumBalance - amount)
                return Abort(transaction, TransactionResult.BalanceLimitExceeded);

            connection.Execute(
                DebitSql,
                new { AccountNumber = senderNumber, Amount = amount },
                transaction);

            connection.Execute(
                CreditSql,
                new { AccountNumber = recipientNumber, Amount = amount },
                transaction);

            DateTime date = DateTime.Now;

            // Names come from the locked rows, so the wording is applied here.
            InsertTransaction(
                connection,
                transaction,
                senderNumber,
                TransactionType.TransferOut,
                amount,
                date,
                TransactionDescriptions.TransferTo(recipient.Name, recipientNumber),
                reference);

            InsertTransaction(
                connection,
                transaction,
                recipientNumber,
                TransactionType.TransferIn,
                amount,
                date,
                TransactionDescriptions.TransferFrom(sender.Name, senderNumber),
                reference);

            transaction.Commit();
            return TransactionResult.Success;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public bool SetFrozenStatus(string accountNumber, bool freeze)
    {
        using var connection = _connectionFactory.CreateConnection();

        int updated = connection.Execute(
            """
            UPDATE Accounts
            SET IsFrozen = @Freeze
            WHERE AccountNumber = @AccountNumber
              AND IsFrozen <> @Freeze;
            """,
            new
            {
                AccountNumber = accountNumber,
                Freeze = freeze
            });

        return updated == 1;
    }

    public IReadOnlyList<LegacyPin> GetLegacyPins()
    {
        using var connection = _connectionFactory.CreateConnection();

        return connection.Query<LegacyPin>(
            """
            SELECT AccountNumber, PinHash AS StoredValue
            FROM Accounts
            WHERE PinHash NOT LIKE 'PBKDF2$%';
            """).ToList();
    }

    public bool ReplacePinHash(
        string accountNumber,
        string currentValue,
        string newHash)
    {
        using var connection = _connectionFactory.CreateConnection();

        int updated = connection.Execute(
            """
            UPDATE Accounts
            SET PinHash = @NewHash
            WHERE AccountNumber = @AccountNumber
              AND PinHash = @CurrentValue;
            """,
            new
            {
                AccountNumber = accountNumber,
                CurrentValue = currentValue,
                NewHash = newHash
            });

        return updated == 1;
    }
    public PinAttemptOutcome VerifyPinWithLockout(string accountNumber, string pin)
    {
        using var connection = _connectionFactory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            // The row lock makes parallel guesses queue up, so none can dodge the counter.
            PinStateRow? row = connection.QuerySingleOrDefault<PinStateRow>(
                """
                SELECT PinHash,
                       CASE WHEN LockedUntil IS NOT NULL
                                 AND LockedUntil > UTC_TIMESTAMP()
                            THEN TIMESTAMPDIFF(SECOND, UTC_TIMESTAMP(), LockedUntil)
                            ELSE 0 END AS LockSecondsRemaining
                FROM Accounts
                WHERE AccountNumber = @AccountNumber
                FOR UPDATE;
                """,
                new { AccountNumber = accountNumber },
                transaction);

            if (row == null)
            {
                transaction.Rollback();
                return PinAttemptOutcome.AccountNotFound;
            }

            // Locked: refuse without checking the PIN and without counting the attempt.
            if (row.LockSecondsRemaining > 0)
            {
                transaction.Rollback();
                return PinAttemptOutcome.Locked;
            }

            if (_pinHasher.Verify(pin, row.PinHash))
            {
                connection.Execute(
                    """
                    UPDATE Accounts
                    SET FailedPinAttempts = 0,
                        LastFailedPinAt = NULL,
                        LockedUntil = NULL
                    WHERE AccountNumber = @AccountNumber
                      AND (FailedPinAttempts <> 0 OR LockedUntil IS NOT NULL);
                    """,
                    new { AccountNumber = accountNumber },
                    transaction);

                transaction.Commit();
                return PinAttemptOutcome.Verified;
            }

            RecordFailedPin(connection, transaction, accountNumber);

            // Commit, not roll back: the failure count must be kept.
            transaction.Commit();
            return PinAttemptOutcome.WrongPin;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public int GetLockSecondsRemaining(string accountNumber)
    {
        using var connection = _connectionFactory.CreateConnection();

        return connection.QuerySingleOrDefault<int>(
            """
            SELECT CASE WHEN LockedUntil IS NOT NULL
                             AND LockedUntil > UTC_TIMESTAMP()
                        THEN TIMESTAMPDIFF(SECOND, UTC_TIMESTAMP(), LockedUntil)
                        ELSE 0 END
            FROM Accounts
            WHERE AccountNumber = @AccountNumber;
            """,
            new { AccountNumber = accountNumber });
    }

    /// <summary>Counts a wrong PIN (restarting after 24 quiet hours) and locks on every 5th.</summary>
    private static void RecordFailedPin(
        SqlConnection connection,
        SqlTransaction transaction,
        string accountNumber)
    {
        // MySQL applies SET left to right, so the first line still sees the old LastFailedPinAt.
        connection.Execute(
            """
            UPDATE Accounts
            SET FailedPinAttempts =
                    IF(LastFailedPinAt IS NULL
                       OR LastFailedPinAt < UTC_TIMESTAMP() - INTERVAL @ResetHours HOUR,
                       1,
                       FailedPinAttempts + 1),
                LastFailedPinAt = UTC_TIMESTAMP()
            WHERE AccountNumber = @AccountNumber;
            """,
            new
            {
                AccountNumber = accountNumber,
                ResetHours = PinLockoutPolicy.ResetAfterHours
            },
            transaction);

        int attempts = connection.QuerySingle<int>(
            """
            SELECT FailedPinAttempts
            FROM Accounts
            WHERE AccountNumber = @AccountNumber;
            """,
            new { AccountNumber = accountNumber },
            transaction);

        int? lockSeconds = PinLockoutPolicy.LockSeconds(attempts);

        if (lockSeconds is null)
            return;

        connection.Execute(
            """
            UPDATE Accounts
            SET LockedUntil = DATE_ADD(UTC_TIMESTAMP(), INTERVAL @Seconds SECOND)
            WHERE AccountNumber = @AccountNumber;
            """,
            new { AccountNumber = accountNumber, Seconds = lockSeconds.Value },
            transaction);
    }

    private static LockedAccountRow? LockAccount(
        SqlConnection connection,
        SqlTransaction transaction,
        string accountNumber)
    {
        return connection.QuerySingleOrDefault<LockedAccountRow>(
            """
            SELECT AccountNumber, Name, PinHash, Balance, IsFrozen
            FROM Accounts
            WHERE AccountNumber = @AccountNumber
            FOR UPDATE;
            """,
            new { AccountNumber = accountNumber },
            transaction);
    }

    private static void InsertTransaction(
        SqlConnection connection,
        SqlTransaction transaction,
        string accountNumber,
        TransactionType type,
        decimal amount,
        DateTime date,
        string description,
        string reference)
    {
        connection.Execute(
            InsertTransactionSql,
            new
            {
                AccountNumber = accountNumber,
                Type = type.ToString(),
                Amount = amount,
                Date = date,
                Description = description,
                Reference = reference
            },
            transaction);
    }

    private static TransactionResult Abort(
        SqlTransaction transaction,
        TransactionResult result)
    {
        transaction.Rollback();
        return result;
    }

    private sealed class LockedAccountRow
    {
        public string AccountNumber { get; set; } = "";
        public string Name { get; set; } = "";
        public string PinHash { get; set; } = "";
        public decimal Balance { get; set; }
        public bool IsFrozen { get; set; }
    }

    private sealed class AccountRow
    {
        public string AccountNumber { get; set; } = "";
        public string Name { get; set; } = "";
        public string Branch { get; set; } = "";
        public string Phone { get; set; } = "";
        public string PinHash { get; set; } = "";
        public decimal Balance { get; set; }
        public bool IsFrozen { get; set; }
    }

    private sealed class TransactionRow
    {
        public string Type { get; set; } = "";
        public decimal Amount { get; set; }
        public DateTime Date { get; set; }
        public string Description { get; set; } = "";
        public string? Reference { get; set; }
    }
    private sealed class PinStateRow
    {
        public string PinHash { get; set; } = "";
        public long LockSecondsRemaining { get; set; }
    }
}

using Dapper;
using TatumWelcome2.Domain.Interfaces;
using TatumWelcome2.Domain.Models;
using TatumWelcome2.Infrastructure.Database;

namespace TatumWelcome2.Infrastructure.Repositories;

public sealed class IncomingInterbankTransferRepository : IIncomingInterbankTransferRepository
{
    private const string SelectTransferSql = """
        SELECT Reference, SenderAccountNumber,
               RecipientAccountNumber, Amount, Status
        FROM IncomingInterbankTransfers
        WHERE Reference = @Reference;
        """;

    private readonly IDbConnectionFactory _factory;

    public IncomingInterbankTransferRepository(IDbConnectionFactory factory)
    {
        _factory = factory;
    }

    public IncomingTransferWrite ReceiveTransfer(
        InterbankTransferRequestDto request,
        string description)
    {
        using var connection = _factory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            // Reserve the reference (0 rows when it already exists).
            int inserted = connection.Execute(
                """
                INSERT IGNORE INTO IncomingInterbankTransfers
                    (Reference, SenderAccountNumber,
                     RecipientAccountNumber, Amount, Status)
                VALUES
                    (@Reference, @SenderAccountNumber,
                     @RecipientAccountNumber, @Amount, 'Processing');
                """,
                request,
                transaction);

            if (inserted == 0)
            {
                IncomingTransferOutcome? existing =
                    connection.QuerySingleOrDefault<IncomingTransferOutcome>(
                        SelectTransferSql,
                        new { request.Reference },
                        transaction);

                transaction.Rollback();

                return new IncomingTransferWrite(
                    IncomingTransferWriteStatus.AlreadyExists,
                    existing);
            }

            RecipientRow? recipient = connection.QuerySingleOrDefault<RecipientRow>(
                """
                SELECT Balance
                FROM Accounts
                WHERE AccountNumber = @RecipientAccountNumber
                FOR UPDATE;
                """,
                new { request.RecipientAccountNumber },
                transaction);

            if (recipient == null ||
                recipient.Balance > AccountLimits.MaximumBalance - request.Amount)
            {
                connection.Execute(
                    """
                    UPDATE IncomingInterbankTransfers
                    SET Status = 'Rejected'
                    WHERE Reference = @Reference;
                    """,
                    new { request.Reference },
                    transaction);

                transaction.Commit();

                return new IncomingTransferWrite(
                    IncomingTransferWriteStatus.RecipientRejected);
            }

            connection.Execute(
                """
                UPDATE Accounts
                SET Balance = Balance + @Amount
                WHERE AccountNumber = @RecipientAccountNumber;
                """,
                request,
                transaction);

            connection.Execute(
                """
                INSERT INTO Transactions
                    (AccountNumber, Type, Amount, Date,
                     Description, Reference)
                VALUES
                    (@RecipientAccountNumber, 'TransferIn',
                     @Amount, @Date, @Description, @Reference);
                """,
                new
                {
                    request.RecipientAccountNumber,
                    request.Amount,
                    Date = DateTime.UtcNow,
                    Description = description,
                    request.Reference
                },
                transaction);

            connection.Execute(
                """
                UPDATE IncomingInterbankTransfers
                SET Status = 'Successful'
                WHERE Reference = @Reference;
                """,
                new { request.Reference },
                transaction);

            transaction.Commit();

            return new IncomingTransferWrite(IncomingTransferWriteStatus.Credited);
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public IncomingTransferOutcome? GetOutcome(string reference)
    {
        using var connection = _factory.CreateConnection();

        return connection.QuerySingleOrDefault<IncomingTransferOutcome>(
            SelectTransferSql,
            new { Reference = reference });
    }

    private sealed class RecipientRow
    {
        public decimal Balance { get; set; }
    }
}

using Dapper;
using Microsoft.Data.SqlClient;
using TatumWelcome2.BLL.DTOs;
using TatumWelcome2.BLL.IServices;
using TatumWelcome2.Domain.Interfaces;
using TatumWelcome2.Domain.Models;
using TatumWelcome2.Infrastructure.Database;

namespace TatumWelcome2.Infrastructure.Repositories;

public sealed class OutgoingBillPaymentRepository : IOutgoingBillPaymentRepository
{
    private const string SelectPaymentSql = """
        SELECT Reference, AccountNumber, BankId, PaymentType, Network,
               PhoneNumber, Amount, DataPlanName, Description,
               Status, CreatedAt
        FROM OutgoingBillPayments
        WHERE Reference = @Reference
        """;

    private readonly IDbConnectionFactory _factory;
    private readonly IPinHasher _pinHasher;

    public OutgoingBillPaymentRepository(
        IDbConnectionFactory factory,
        IPinHasher pinHasher)
    {
        _factory = factory;
        _pinHasher = pinHasher;
    }

    public OutgoingReservation Reserve(
        BillPaymentRequest request,
        string pin,
        string description)
    {
        using var connection = _factory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            AccountRow? account = connection.QuerySingleOrDefault<AccountRow>(
                """
                SELECT PinHash, Balance, IsFrozen
                FROM Accounts
                WHERE AccountNumber = @AccountNumber
                FOR UPDATE;
                """,
                new { request.AccountNumber },
                transaction);

            if (account == null)
                return Reject(transaction, TransactionResult.AccountNotFound);

            if (!_pinHasher.Verify(pin, account.PinHash))
                return Reject(transaction, TransactionResult.InvalidPin);

            OutgoingBillPaymentRecord? existing =
                connection.QuerySingleOrDefault<OutgoingBillPaymentRecord>(
                    SelectPaymentSql,
                    new { request.Reference },
                    transaction);

            if (existing != null)
            {
                transaction.Rollback();

                return new OutgoingReservation(
                    OutgoingReservationStatus.Existing,
                    Existing: existing);
            }

            if (account.IsFrozen)
                return Reject(transaction, TransactionResult.AccountFrozen);

            if (account.Balance < request.Amount)
                return Reject(transaction, TransactionResult.InsufficientFunds);

            connection.Execute(
                """
                UPDATE Accounts
                SET Balance = Balance - @Amount
                WHERE AccountNumber = @AccountNumber;
                """,
                new { request.Amount, request.AccountNumber },
                transaction);

            connection.Execute(
                """
                INSERT INTO OutgoingBillPayments
                    (Reference, AccountNumber, BankId, PaymentType,
                     Network, PhoneNumber, Amount, DataPlanName,
                     Description, Status)
                VALUES
                    (@Reference, @AccountNumber, @BankId, @PaymentType,
                     @Network, @PhoneNumber, @Amount, @DataPlanName,
                     @Description, 'Pending');
                """,
                new
                {
                    request.Reference,
                    request.AccountNumber,
                    request.BankId,
                    request.PaymentType,
                    request.Network,
                    request.PhoneNumber,
                    request.Amount,
                    request.DataPlanName,
                    Description = description
                },
                transaction);

            transaction.Commit();
            return new OutgoingReservation(OutgoingReservationStatus.Reserved);
        }
        catch (SqlException ex) when (SqlServerErrors.IsUniqueViolation(ex))
        {
            // Same reference taken by another account at the same moment.
            transaction.Rollback();

            return new OutgoingReservation(
                OutgoingReservationStatus.Existing,
                Existing: GetByReference(request.Reference));
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public bool Complete(string reference, bool success)
    {
        using var connection = _factory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            OutgoingBillPaymentRecord? payment =
                connection.QuerySingleOrDefault<OutgoingBillPaymentRecord>(
                    SelectPaymentSql + "\nFOR UPDATE;",
                    new { Reference = reference },
                    transaction);

            if (payment == null)
            {
                transaction.Rollback();
                return false;
            }

            // Never complete the same payment twice.
            if (payment.Status != OutgoingBillPaymentStatus.Pending)
            {
                transaction.Rollback();

                return payment.Status == (success
                    ? OutgoingBillPaymentStatus.Success
                    : OutgoingBillPaymentStatus.Failed);
            }

            if (success)
            {
                string type = payment.PaymentType == BillPaymentTypes.Data
                    ? TransactionType.DataPurchase.ToString()
                    : TransactionType.AirtimePurchase.ToString();

                connection.Execute(
                    """
                    INSERT INTO Transactions
                        (AccountNumber, Type, Amount, Date,
                         Description, Reference)
                    VALUES
                        (@AccountNumber, @Type, @Amount, @Date,
                         @Description, @Reference);
                    """,
                    new
                    {
                        payment.AccountNumber,
                        Type = type,
                        payment.Amount,
                        Date = DateTime.UtcNow,
                        payment.Description,
                        payment.Reference
                    },
                    transaction);
            }
            else
            {
                decimal balance = connection.QuerySingle<decimal>(
                    """
                    SELECT Balance
                    FROM Accounts
                    WHERE AccountNumber = @AccountNumber
                    FOR UPDATE;
                    """,
                    new { payment.AccountNumber },
                    transaction);

                // A refund must not push the account over its balance cap.
                if (balance > AccountLimits.MaximumBalance - payment.Amount)
                {
                    transaction.Rollback();
                    return false;
                }

                connection.Execute(
                    """
                    UPDATE Accounts
                    SET Balance = Balance + @Amount
                    WHERE AccountNumber = @AccountNumber;
                    """,
                    new { payment.AccountNumber, payment.Amount },
                    transaction);
            }

            connection.Execute(
                """
                UPDATE OutgoingBillPayments
                SET Status = @Status
                WHERE Reference = @Reference;
                """,
                new
                {
                    Reference = reference,
                    Status = success
                        ? OutgoingBillPaymentStatus.Success
                        : OutgoingBillPaymentStatus.Failed
                },
                transaction);

            transaction.Commit();
            return true;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public bool MarkReferenceConflict(string reference)
    {
        using var connection = _factory.CreateConnection();

        int updated = connection.Execute(
            """
            UPDATE OutgoingBillPayments
            SET Status = 'ReferenceConflict'
            WHERE Reference = @Reference
              AND Status = 'Pending';
            """,
            new { Reference = reference });

        return updated == 1;
    }

    public OutgoingBillPaymentRecord? GetByReference(string reference)
    {
        using var connection = _factory.CreateConnection();

        return connection.QuerySingleOrDefault<OutgoingBillPaymentRecord>(
            SelectPaymentSql + ";",
            new { Reference = reference });
    }

    public List<OutgoingBillPaymentRecord> GetPending()
    {
        using var connection = _factory.CreateConnection();

        return connection.Query<OutgoingBillPaymentRecord>(
            """
            SELECT Reference, AccountNumber, BankId, PaymentType, Network,

    private readonly IPinHasher _pinHasher;

    public OutgoingInterbankTransferRepository(

        _pinHasher = pinHasher;

    public InterbankOutgoingTransferRecord? GetByReference(string reference)

    public TransactionResult DebitAndCreate(
        string senderNumber,
        string recipientNumber,
        string bankUrl,
        decimal amount,
        string pin,
        string reference,
        string description)
    {
        using var connection = _factory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            SenderRow? sender = connection.QuerySingleOrDefault<SenderRow>(
                """
                SELECT PinHash, Balance, IsFrozen
                FROM Accounts
                WHERE AccountNumber = @SenderNumber
                FOR UPDATE;
                """,
                new { SenderNumber = senderNumber },
                transaction);

            if (sender == null)
                return Abort(transaction, TransactionResult.AccountNotFound);

            string? existing = connection.QuerySingleOrDefault<string>(
                """
                SELECT Reference
                FROM OutgoingInterbankTransfers
                WHERE Reference = @Reference;
                """,
                new { Reference = reference },
                transaction);

            if (existing != null)
                return Abort(transaction, TransactionResult.DuplicateReference);

            if (!_pinHasher.Verify(pin, sender.PinHash))
                return Abort(transaction, TransactionResult.InvalidPin);

            if (sender.IsFrozen)
                return Abort(transaction, TransactionResult.AccountFrozen);

            if (sender.Balance < amount)
                return Abort(transaction, TransactionResult.InsufficientFunds);

            connection.Execute(
                """
                UPDATE Accounts
                SET Balance = Balance - @Amount
                WHERE AccountNumber = @SenderNumber;
                """,
                new { SenderNumber = senderNumber, Amount = amount },
                transaction);

            connection.Execute(
                """
                INSERT INTO Transactions
                    (AccountNumber, Type, Amount, Date,
                     Description, Reference)
                VALUES
                    (@SenderNumber, 'TransferOut', @Amount,
                     @Date, @Description, @Reference);
                """,
                new
                {
                    SenderNumber = senderNumber,
                    Amount = amount,
                    Date = DateTime.UtcNow,
                    Description = description,
                    Reference = reference
                },
                transaction);

            connection.Execute(
                """
                INSERT INTO OutgoingInterbankTransfers
                    (Reference, SenderAccountNumber,
                     RecipientAccountNumber, BankUrl,
                     Amount, Status)
                VALUES
                    (@Reference, @SenderNumber,
                     @RecipientNumber, @BankUrl,
                     @Amount, 'Pending');
                """,
                new
                {
                    Reference = reference,
                    SenderNumber = senderNumber,
                    RecipientNumber = recipientNumber,
                    BankUrl = bankUrl,
                    Amount = amount
                },
                transaction);

            transaction.Commit();
            return TransactionResult.Success;
        }
        catch (SqlException ex) when (SqlServerErrors.IsUniqueViolation(ex))
        {
            transaction.Rollback();
            return TransactionResult.DuplicateReference;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

    public List<InterbankOutgoingTransferRecord> GetPendingTransfers(
        WorkPartition partition)
    {
        partition.EnsureValid();

        using var connection = _factory.CreateConnection();

        // CRC32 is computed by MySQL, so every instance agrees on
        // ownership (string.GetHashCode differs per process).
        return connection.Query<InterbankOutgoingTransferRecord>(
            """
            SELECT Reference, SenderAccountNumber,
                   RecipientAccountNumber, BankUrl,
                   Amount, Status
            FROM OutgoingInterbankTransfers
            WHERE Status = 'Pending'
              AND MOD(CRC32(Reference), @PartitionCount) = @PartitionIndex
            ORDER BY CreatedAt;
            """,
            new
            {
                PartitionIndex = partition.Index,
                PartitionCount = partition.Count
            }).ToList();
    }

    public bool Reverse(string reference, string description)
    {
        using var connection = _factory.CreateConnection();
        connection.Open();

        using var transaction = connection.BeginTransaction();

        try
        {
            InterbankOutgoingTransferRecord? transfer =
                connection.QuerySingleOrDefault<InterbankOutgoingTransferRecord>(
                    """
                    SELECT Reference, SenderAccountNumber,
                           RecipientAccountNumber, BankUrl,
                           Amount, Status
                    FROM OutgoingInterbankTransfers
                    WHERE Reference = @Reference
                    FOR UPDATE;
                    """,
                    new { Reference = reference },
                    transaction);

            if (transfer == null ||
                transfer.Status != InterbankTransferStatus.Pending)
            {
                transaction.Rollback();
                return false;
            }

            decimal balance = connection.QuerySingle<decimal>(
                """
                SELECT Balance
                FROM Accounts
                WHERE AccountNumber = @AccountNumber
                FOR UPDATE;
                """,
                new { AccountNumber = transfer.SenderAccountNumber },
                transaction);

            if (balance > AccountLimits.MaximumBalance - transfer.Amount)
            {
                transaction.Rollback();
                return false;
            }

            connection.Execute(
                """
                UPDATE Accounts
                SET Balance = Balance + @Amount
                WHERE AccountNumber = @AccountNumber;
                """,
                new
                {
                    AccountNumber = transfer.SenderAccountNumber,
                    transfer.Amount
                },
                transaction);

            connection.Execute(
                """
                INSERT INTO Transactions
                    (AccountNumber, Type, Amount, Date,
                     Description, Reference)
                VALUES
                    (@AccountNumber, 'TransferIn', @Amount,
                     @Date, @Description, @Reference);
                """,
                new
                {
                    AccountNumber = transfer.SenderAccountNumber,
                    transfer.Amount,
                    Date = DateTime.UtcNow,
                    Description = description,
                    Reference = reference
                },
                transaction);

            connection.Execute(
                """
                UPDATE OutgoingInterbankTransfers
                SET Status = 'Reversed'
                WHERE Reference = @Reference;
                """,
                new { Reference = reference },
                transaction);

            transaction.Commit();
            return true;
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }

        TransactionResult result)

        return result;
